  3480→    #  SSE auto-reconnect with exponential backoff
  3481→    # ------------------------------------------------------------------ #
  3482→    def stream_sync_updates_resilient(
  3483→        self,
  3484→        entity_type: str,
  3485→        entity_id: str,
  3486→        *,
  3487→        max_retries: int = 5,
  3488→        base_backoff: float = 1.0,
  3489→        max_backoff: float = 30.0,
  3490→    ) -> Iterator[dict]:
  3491→        """Stream sync updates with auto-reconnect and exponential backoff.
  3492→
  3493→        Unlike ``stream_sync_updates()`` (single connection), this method
  3494→        automatically reconnects on failure with exponential backoff,
  3495→        matching the Vibes UI's behavior.
  3496→
  3497→        Parameters
  3498→        ----------
  3499→        entity_type, entity_id : str
  3500→            Entity to watch.
  3501→        max_retries : int
  3502→            Max consecutive failures before giving up (default 5).
  3503→        base_backoff : float
  3504→            Initial backoff in seconds (default 1.0).
  3505→        max_backoff : float
  3506→            Cap on backoff (default 30.0).
  3507→
  3508→        Yields
  3509→        ------
  3510→        dict
  3511→            Same events as ``stream_sync_updates()``.
  3512→        """
  3513→        retries = 0
  3514→        while retries < max_retries:
  3515→            try:
  3516→                for event in self.stream_sync_updates(entity_type, entity_id):
  3517→                    retries = 0  # reset on successful event
  3518→                    yield event
  3519→            except (VibesAPIError, ConnectionError) as e:
  3520→                retries += 1
  3521→                if retries >= max_retries:
  3522→                    raise
  3523→                backoff = min(base_backoff * (2 ** retries), max_backoff)
  3524→                time.sleep(backoff)
  3525→
  3526→    def stream_batch_updates_resilient(
  3527→        self,
  3528→        batch_id: str,
  3529→        *,
  3530→        max_retries: int = 5,
  3531→        base_backoff: float = 1.0,
  3532→        max_backoff: float = 30.0,
  3533→        idle_timeout: float = 300.0,
  3534→    ) -> Iterator[dict]:
  3535→        """Stream batch updates with auto-reconnect.
  3536→
  3537→        Stops when the batch is complete (``isComplete: true``) or after
  3538→        ``idle_timeout`` seconds with no events.
  3539→        """
  3540→        retries = 0
  3541→        last_event = time.time()
  3542→        while retries < max_retries and time.time() - last_event < idle_timeout:
  3543→            try:
  3544→                for event in self.stream_batch_updates(batch_id):
  3545→                    last_event = time.time()
  3546→                    retries = 0
  3547→                    yield event
  3548→                    if event.get("isComplete"):
  3549→                        return
  3550→            except (VibesAPIError, ConnectionError):
  3551→                retries += 1
  3552→                if retries >= max_retries:
  3553→                    raise
  3554→                backoff = min(base_backoff * (2 ** retries), max_backoff)
  3555→                time.sleep(backoff)

  3995→    # ------------------------------------------------------------------ #
  3996→    #  Midjourney parameter parsing (sref / oref / chaos / stylize / etc.)
  3997→    # ------------------------------------------------------------------ #
  3998→    @staticmethod
  3999→    def parse_midjourney_params(prompt: str) -> dict:
  4000→        """Parse Midjourney-style parameters from a prompt.
  4001→
  4002→        Extracts parameters like ``--sref``, ``--oref``, ``--sw``,
  4003→        ``--ow``, ``--seed``, ``--chaos``, ``--stylize``, ``--ar``,
  4004→        ``--v``, ``--niji``, ``--loop``, ``--raw``, ``--tile``,
  4005→        ``--turbo``, ``--relax``, ``--stealth``, ``--public``,
  4006→        ``--draft``, ``--video``, ``--fast``, ``--style``.
  4007→
  4008→        Returns
  4009→        -------
  4010→        dict
  4011→            ``{cleanPrompt: str, parameters: {sref: ..., oref: ...,
  4012→            sref_weight: ..., oref_weight: ..., seed: ..., chaos: ...,
  4013→            stylize: ..., aspect_ratio: ..., version: ..., niji: bool,
  4014→            loop: bool, raw: bool, tile: bool, ...}}``
  4015→        """
  4016→        params: Dict[str, Any] = {}
  4017→        clean = prompt
  4018→
  4019→        # --sref (style reference, can be URL, random, or numeric IDs)
  4020→        import re
  4021→        sref_match = re.search(r'--sref\s+([^\s-]+(?:\s+[^\s-]+)*?)(?=\s+--|$)', prompt)
  4022→        if sref_match:
  4023→            sref_val = sref_match.group(1).strip()
  4024→            if sref_val.lower() == "random":
  4025→                params["sref_random"] = True
  4026→                params["sref_value"] = sref_val
  4027→            else:
  4028→                params["sref_value"] = sref_val
  4029→                # Extract numeric IDs
  4030→                nums = [x for x in sref_val.split() if x.isdigit()]
  4031→                if nums:
  4032→                    params["sref_ids"] = [int(n) for n in nums]
  4033→            clean = clean.replace(sref_match.group(0), "").strip()
  4034→
  4035→        # --oref (object/character reference)
  4036→        oref_match = re.search(r'--oref\s+([^\s-]+(?:\s+[^\s-]+)*?)(?=\s+--|$)', prompt)
  4037→        if oref_match:
  4038→            oref_val = oref_match.group(1).strip()
  4039→            params["oref_value"] = oref_val
  4040→            nums = [x for x in oref_val.split() if x.isdigit()]
  4041→            if nums:
  4042→                params["oref_ids"] = [int(n) for n in nums]
  4043→            clean = clean.replace(oref_match.group(0), "").strip()
  4044→
  4045→        # --sw (sref weight)
  4046→        sw_match = re.search(r'--sw\s+([\d.]+)', prompt)
  4047→        if sw_match:
  4048→            params["sref_weight"] = float(sw_match.group(1))
  4049→            clean = clean.replace(sw_match.group(0), "").strip()
  4050→
  4051→        # --ow (oref weight)
  4052→        ow_match = re.search(r'--ow\s+([\d.]+)', prompt)
  4053→        if ow_match:
  4054→            params["oref_weight"] = float(ow_match.group(1))
  4055→            clean = clean.replace(ow_match.group(0), "").strip()
  4056→
  4057→        # --seed
  4058→        seed_match = re.search(r'--seed\s+(\d+)', prompt)
  4059→        if seed_match:
  4060→            params["seed"] = int(seed_match.group(1))
  4061→            clean = clean.replace(seed_match.group(0), "").strip()
  4062→
  4063→        # --chaos / --c
  4064→        chaos_match = re.search(r'--(?:chaos|c)\s+(\d+)', prompt)
  4065→        if chaos_match:
  4066→            params["chaos"] = int(chaos_match.group(1))
  4067→            clean = clean.replace(chaos_match.group(0), "").strip()
  4068→
  4069→        # --stylize / --s

  4071→        if stylize_match:
  4072→            params["stylize"] = int(stylize_match.group(1))
  4073→            clean = clean.replace(stylize_match.group(0), "").strip()
  4074→
  4075→        # --ar (aspect ratio)
  4076→        ar_match = re.search(r'--ar\s+(\d+:\d+)', prompt)
  4077→        if ar_match:
  4078→            params["aspect_ratio"] = ar_match.group(1)

  4080→
  4081→        # --v (version)

  4083→        if v_match:

  4085→            clean = clean.replace(v_match.group(0), "").strip()
  4086→
  4087→        # Boolean flags
  4088→        for flag in ["niji", "loop", "raw", "tile", "turbo", "relax",
  4089→                     "stealth", "public", "draft", "video", "fast"]:
  4090→            if re.search(rf'--{flag}\b', prompt):
  4091→                params[flag] = True
  4092→                clean = re.sub(rf'--{flag}\b\s*', '', clean).strip()
  4093→
  4094→        # --style
