  2088→    def _download(self, content_item_id: str, output_path: str, endpoint: str) -> str:
  2089→        url = self._url(f"{endpoint}?id={content_item_id}")
  2090→        with self.session.get(url, stream=True, timeout=max(self.timeout, 600)) as r:
  2091→            if not r.ok:
  2092→                raise VibesAPIError(
  2093→                    f"Download failed: HTTP {r.status_code}",
  2094→                    status=r.status_code,
  2095→                )
  2096→            with open(output_path, "wb") as f:
  2097→                for chunk in r.iter_content(chunk_size=64 * 1024):
  2098→                    if chunk:
  2099→                        f.write(chunk)
  2100→        return output_path
